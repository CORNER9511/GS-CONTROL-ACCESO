  <script>

    /* ---------------------------------------------------
       Variables globales
    --------------------------------------------------- */
    let stream;

    const video       = document.getElementById("video");
    const canvas      = document.getElementById("canvas");
    const ctx         = canvas.getContext("2d");
    const statusEl    = document.getElementById("status");

    /* ---------------------------------------------------
       FUNCIÓN: Iniciar cámara del dispositivo
    --------------------------------------------------- */
    async function startCamera() {
      try {
        stream = await navigator.mediaDevices.getUserMedia({ video: true });
        video.srcObject = stream;
      } catch (err) {
        showToast("Error al acceder a la cámara", "error");
      }
    }

    /* Detener cámara */
    function stopCamera() {
      if (stream) {
        stream.getTracks().forEach(t => t.stop());
      }
    }

    startCamera();

    /* ---------------------------------------------------
       Cargar listas desde Google Apps Script
       (Empresas, Sedes, Operaciones)
    --------------------------------------------------- */
    google.script.run
      .withSuccessHandler(data => {

        const sedeSel    = document.getElementById("sede");
        const operSel    = document.getElementById("operacion");
        const empresaSel = document.getElementById("empresa");

        // Se limpian las opciones existentes
        sedeSel.innerHTML    = '<option value="">Seleccione una sede</option>';
        operSel.innerHTML    = '<option value="">Seleccione una operación</option>';
        empresaSel.innerHTML = '<option value="">Seleccione una empresa</option>';

        // Se agregan los valores desde la hoja Config
        data.sedes.forEach(s       => sedeSel.add(new Option(s, s)));
        data.operaciones.forEach(o => operSel.add(new Option(o, o)));
        data.empresas.forEach(e     => empresaSel.add(new Option(e, e)));

      })
      .getSedeOperacionEmpresaList();

    /* ---------------------------------------------------
       Notificaciones tipo toast
    --------------------------------------------------- */
    function showToast(message, type = "success") {
      const container = document.getElementById("toastContainer");
      const toast = document.createElement("div");

      toast.className = `toast ${type}`;
      toast.textContent = message;

      container.appendChild(toast);

      // Remover después de terminar animaciones
      setTimeout(() => toast.remove(), 4000);
    }

    /* ---------------------------------------------------
       Capturar foto y registrar entrada del contratista
    --------------------------------------------------- */
    function takePhotoAndSave(type) {

      const cedulas   = document.getElementById("cedula").value.trim();
      const nombre    = document.getElementById("nombre").value.trim();
      const empresa   = document.getElementById("empresa").value;
      const sede      = document.getElementById("sede").value;
      const operacion = document.getElementById("operacion").value;
      const obsEntrada = document.getElementById("obsEntrada").value.trim();
      const btnEntrada = document.getElementById("btnEntrada");

      // Validación básica
      if (!cedulas || !nombre || !empresa || !sede || !operacion) {
        showToast("Complete todos los campos requeridos", "warning");
        return;
      }

      const listaCedulas = cedulas.split(/[\s,]+/).filter(Boolean);

      // Obtener ubicación del dispositivo
      navigator.geolocation.getCurrentPosition(pos => {

        // Capturar foto
        canvas.width  = video.videoWidth;
        canvas.height = video.videoHeight;
        ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

        const photo = canvas.toDataURL("image/jpeg");

        statusEl.innerText = "Registrando entrada...";
        btnEntrada.disabled = true;

        // Procesar cada cédula
        listaCedulas.forEach(id => {

          const payload = {
            cedula: id,
            nombre: nombre,
            type: type,
            lat: pos.coords.latitude,
            lng: pos.coords.longitude,
            photoDataURL: photo,
            empresa: empresa,
            sede: sede,
            operacion: operacion,
            comentarios: obsEntrada
          };

          google.script.run
            .withSuccessHandler(res => {
              if (res.success) {
                showToast(res.message, "success");
                statusEl.innerText = res.message;

                setTimeout(() => {
                  stopCamera();
                  goBack();
                }, 2500);

              } else {
                showToast(res.message, "warning");
                statusEl.innerText = res.message;
                btnEntrada.disabled = false;
              }
            })
            .withFailureHandler(err => {
              showToast("Error: " + err.message, "error");
              btnEntrada.disabled = false;
            })
            .saveContractorAttendance(payload);

        });

      }, () => {
        showToast("No se pudo obtener la ubicación", "warning");
      });

    }

    /* ---------------------------------------------------
       Regresar al menú principal
    --------------------------------------------------- */
    function goBack() {
      stopCamera();
      google.script.run.withSuccessHandler(html => {
        document.open();
        document.write(html);
        document.close();
      }).loadView('ContractorMenu');
    }
