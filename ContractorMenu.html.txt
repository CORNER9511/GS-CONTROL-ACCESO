  <div class="card">
    <img id="Logo" alt="Logo">

    <h1>Panel Contratista</h1>
    <p>Seleccione una opción:</p>

    <!-- Botones que navegan a otras vistas -->
    <button onclick="goToView('ContractorEntrada')">Registrar Entrada</button>
    <button onclick="goToView('ContractorSalida')">Registrar Salida</button>
    <button onclick="goToView('')">Volver al Inicio</button>

  <script>
    /* -----------------------------------------------------
       AL CARGAR LA PÁGINA: Manejo del logo desde Drive
       - Primero revisa si hay un logo guardado en caché.
       - Luego intenta obtener uno actualizado desde Drive.
    ------------------------------------------------------- */
    document.addEventListener('DOMContentLoaded', function() {
      const logo = document.getElementById('Logo');
      const cachedLogo = localStorage.getItem('cachedLogo');

      // Mostrar logo guardado en caché inmediatamente (si existe)
      if (cachedLogo) {
        logo.src = cachedLogo;
        logo.style.opacity = 1; // Mostrar logo suavemente
      }

      // Solicitar versión actualizada desde Google Drive
      google.script.run
        .withSuccessHandler(function(imageUrl) {
          if (imageUrl) {
            logo.src = imageUrl;
            localStorage.setItem('cachedLogo', imageUrl); // Actualizar caché
            logo.onload = () => { logo.style.opacity = 1; }; // Transición suave al aparecer
          }
        })
        .getDriveImage();
    });


    /* -----------------------------------------------------
       FUNCIÓN PARA NAVEGAR ENTRE VISTAS DEL WEB APP
       - Cambia el estado de la URL
       - Solicita al servidor la vista correspondiente
    ------------------------------------------------------- */
    function goToView(view) {
      google.script.history.push({
        state: { view: view },
        location: '?view=' + view
      });

      google.script.run
        .withSuccessHandler(loadHtml)
        .loadView(view);
    }

    /* -----------------------------------------------------
       FUNCIÓN: Cargar el HTML recibido desde Apps Script
    ------------------------------------------------------- */
    function loadHtml(html) {
      document.open();
      document.write(html);
      document.close();
    }
  </script>
