  <script>
    /* -----------------------------------------------------
       VARIABLES PRINCIPALES
    ------------------------------------------------------- */
    let stream;
    const video     = document.getElementById("video");
    const canvas    = document.getElementById("canvas");
    const ctx       = canvas.getContext("2d");
    const statusEl  = document.getElementById("status");

    /* -----------------------------------------------------
       INICIO DE LA CÁMARA DEL DISPOSITIVO
    ------------------------------------------------------- */
    async function startCamera() {
      try {
        stream = await navigator.mediaDevices.getUserMedia({ video: true });
        video.srcObject = stream;
      } catch (err) {
        showToast("Error al acceder a la cámara.", "error");
      }
    }

    /* Detener la cámara */
    function stopCamera() {
      if (stream) {
        stream.getTracks().forEach(t => t.stop());
      }
    }

    /* Iniciar cámara al cargar la vista */
    startCamera();

    /* -----------------------------------------------------
       SISTEMA DE NOTIFICACIONES EN PANTALLA (TOAST)
    ------------------------------------------------------- */
    function showToast(message, type = "success") {
      const container = document.getElementById("toastContainer");
      const toast = document.createElement("div");
      toast.className = `toast ${type}`;
      toast.textContent = message;
      container.appendChild(toast);
      setTimeout(() => toast.remove(), 4000);
    }

    /* -----------------------------------------------------
       CAPTURA DE FOTO + ENVÍO DE REGISTRO A APP SCRIPT
    ------------------------------------------------------- */
    function takePhotoAndSave(type) {

      // Obtener valores del formulario
      const cedulas = document.getElementById("cedula").value.trim();
      const obsSalida = document.getElementById("obsSalida").value.trim();
      const btnSalida = document.getElementById("btnSalida");

      if (!cedulas) {
        showToast("Ingrese al menos una cédula.", "warning");
        return;
      }

      // Separar múltiples cédulas
      const listaCedulas = cedulas.split(/[\s,]+/).filter(Boolean);

      // Obtener ubicación GPS
      navigator.geolocation.getCurrentPosition(pos => {

        // Capturar la foto
        canvas.width = video.videoWidth;
        canvas.height = video.videoHeight;
        ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

        const photo = canvas.toDataURL("image/jpeg");

        statusEl.innerText = "Registrando salida...";
        btnSalida.disabled = true;

        // Contadores de resultados
        let exitos = 0;
        let fallos = 0;
        let cedulasError = [];
        let completados = 0;

        // Ejecutar registro por cada cédula
        listaCedulas.forEach(id => {

          const payload = {
            cedula: id,
            type: type,
            lat: pos.coords.latitude,
            lng: pos.coords.longitude,
            photoDataURL: photo,
            comentarios: obsSalida
          };

          google.script.run
            .withSuccessHandler(res => {

              if (res.success) {
                exitos++;
                showToast(id + ": salida registrada correctamente.", "success");
              } else {
                fallos++;
                cedulasError.push(id);
                showToast(id + ": " + res.message, "warning");
              }

              completados++;

              // Evaluar solo cuando termine el proceso completo
              if (completados === listaCedulas.length) {

                btnSalida.disabled = false;

                if (fallos === 0) {
                  statusEl.innerText = exitos + " salidas registradas correctamente.";
                  showToast(exitos + " salidas registradas correctamente.", "success");
                  document.getElementById("cedula").value = "";

                  setTimeout(() => {
                    stopCamera();
                    goBack();
                  }, 2500);

                } else {
                  statusEl.innerText =
                    exitos + " correctas, " + fallos + " con novedad.\n" +
                    "Cédulas con error: " + cedulasError.join(", ");
                  showToast(fallos + " cédulas con error.", "warning");
                }
              }
            })

            .withFailureHandler(err => {
              fallos++;
              cedulasError.push(id);
              completados++;
              showToast(id + ": " + err.message, "error");
            })

            .saveContractorAttendance(payload);
        });
      });
    }

    /* -----------------------------------------------------
       VOLVER AL MENÚ PRINCIPAL DEL CONTRATISTA
    ------------------------------------------------------- */
    function goBack() {
      stopCamera();
      google.script.run.withSuccessHandler(html => {
        document.open();
        document.write(html);
        document.close();
      }).loadView('ContractorMenu');
    }
  </script>
