  <script>

    let stream;
    const video = document.getElementById("video");
    const canvas = document.getElementById("canvas");
    const ctx = canvas.getContext("2d");
    const statusEl = document.getElementById("status");


    /* =========================================================
       INICIAR CÁMARA
       ========================================================= */
    async function startCamera() {
      try {
        stream = await navigator.mediaDevices.getUserMedia({ video: true });
        video.srcObject = stream;

      } catch (err) {
        showToast("Error al acceder a la cámara", "error");
      }
    }


    /* =========================================================
       DETENER CÁMARA
       ========================================================= */
    function stopCamera() {
      if (stream) stream.getTracks().forEach(t => t.stop());
    }


    /* =========================================================
       CARGAR LISTA DE SEDES Y OPERACIONES DESDE APPS SCRIPT
       ========================================================= */
    function loadSedesYOperaciones() {
      google.script.run
        .withSuccessHandler(data => {

          const sedeSel = document.getElementById("sede");
          const operSel = document.getElementById("operacion");

          sedeSel.innerHTML = '<option value="">Seleccione una sede</option>';
          operSel.innerHTML = '<option value="">Seleccione una operación</option>';

          const info = typeof data === "string" ? JSON.parse(data) : data;

          (info.sedes || []).forEach(s => sedeSel.add(new Option(s, s)));
          (info.operaciones || []).forEach(o => operSel.add(new Option(o, o)));
        })

        .withFailureHandler(err => {
          showToast("Error al cargar listas: " + err.message, "error");
        })

        .getSedeOperacionList();
    }


    /* =========================================================
       MOSTRAR TOAST DE NOTIFICACIONES
       ========================================================= */
    function showToast(message, type = "success") {
      const container = document.getElementById("toastContainer");
      const toast = document.createElement("div");

      toast.className = `toast ${type}`;
      toast.textContent = message;

      container.appendChild(toast);

      setTimeout(() => toast.remove(), 4000);
    }


    /* =========================================================
       ACTIVAR CÁMARA Y CARGAR DATOS INICIALES AL CARGAR LA PÁGINA
       ========================================================= */
    window.onload = () => {
      startCamera();
      loadSedesYOperaciones();
    };


    /* =========================================================
       FUNCIÓN PRINCIPAL: REGISTRO DE ENTRADA (MÚLTIPLES CÉDULAS)
       ========================================================= */
    async function takePhotoAndSave(type) {

      // Obtener valores del formulario
      const cedulasInput = document.getElementById("cedulas").value.trim();
      const sede = document.getElementById("sede").value;
      const operacion = document.getElementById("operacion").value;
      const obsEntrada = document.getElementById("obsEntrada").value.trim();

      const btnEntrada = document.getElementById("btnEntrada");
      const progressContainer = document.getElementById("progressContainer");
      const progressBar = document.getElementById("progressBar");

      // Validaciones
      if (!cedulasInput || !sede || !operacion) {
        showToast("Complete todos los campos.", "warning");
        return;
      }

      // Convertir entrada a lista de cédulas
      const listaCedulas = cedulasInput
        .split(/[\n,]+/)
        .map(c => c.trim())
        .filter(c => c !== "");

      if (listaCedulas.length === 0) {
        showToast("Ingrese al menos una cédula válida.", "warning");
        return;
      }

      // Capturar foto del video en el canvas
      canvas.width = video.videoWidth;
      canvas.height = video.videoHeight;
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
      const photo = canvas.toDataURL("image/jpeg");

      // Obtener geolocalización
      let pos;
      try {
        pos = await new Promise((resolve, reject) => {
          navigator.geolocation.getCurrentPosition(resolve, reject, { timeout: 10000 });
        });
      } catch {
        pos = { coords: { latitude: "", longitude: "" } };
        showToast("No se pudo obtener ubicación.", "warning");
      }

      // Construir payload para Apps Script
      const payloadList = listaCedulas.map(id => ({
        user_id: id,
        type: type,
        lat: pos.coords.latitude,
        lng: pos.coords.longitude,
        photoDataURL: photo,
        sede,
        operacion,
        obsEntrada
      }));


      /* ------------------------------------
         MANEJO DE BARRA DE PROGRESO
      ------------------------------------ */
      progressContainer.style.display = "block";
      progressBar.style.width = "0%";

      statusEl.innerText = "Registrando entradas, por favor espere...";
      btnEntrada.disabled = true;

      // Animar barra mientras espera
      let progress = 0;
      const interval = setInterval(() => {
        progress = Math.min(progress + 5, 90);
        progressBar.style.width = progress + "%";
      }, 300);


      /* ------------------------------------
         ENVÍO DE REGISTROS A GOOGLE APPS SCRIPT
      ------------------------------------ */
      google.script.run

        .withSuccessHandler(results => {
          clearInterval(interval);
          progressBar.style.width = "100%";

          const errores = results.filter(r => !r.success);
          const exitos = results.filter(r => r.success);

          if (errores.length > 0) {

            const fallidos = errores.map(e => e.user_id || e.message).join(", ");

            statusEl.innerText =
              `Registros exitosos: ${exitos.length}\n` +
              `Errores: ${errores.length}\n` +
              `Cédulas con error: ${fallidos}`;

            showToast(`Se detectaron ${errores.length} errores. Corrija las cédulas con error.`, "warning");

            btnEntrada.disabled = false;
          }

          else {
            showToast(`${exitos.length} entradas registradas correctamente.`, "success");
            statusEl.innerText = `Todos los registros (${exitos.length}) fueron exitosos.`;

            setTimeout(() => {
              stopCamera();
              goBack();
            }, 2000);
          }
        })

        .withFailureHandler(err => {
          clearInterval(interval);
          progressBar.style.width = "0%";

          showToast("Error: " + err.message, "error");
          statusEl.innerText = "Error general: " + err.message;

          btnEntrada.disabled = false;
        })

        .saveAttendanceBatch(payloadList);
    }


    /* =========================================================
       VOLVER AL MENÚ ANTERIOR
       ========================================================= */
    function goBack() {
      stopCamera();

      google.script.run
        .withSuccessHandler(html => {
          document.open();
          document.write(html);
          document.close();
        })
        .loadView('EmployeeMenu');
    }
