  <script>

    /* ---------------------------------------------------
       CARGA DEL LOGO DESDE CACHÉ Y DESDE GOOGLE DRIVE
    ---------------------------------------------------- */
    document.addEventListener('DOMContentLoaded', function () {

      const logo = document.getElementById('Logo');
      const cachedLogo = localStorage.getItem('cachedLogo');

      // Si hay logo guardado en caché, mostrarlo inmediatamente
      if (cachedLogo) {
        logo.src = cachedLogo;
        logo.style.opacity = 1;
      }

      // Solicitar logo actualizado desde Apps Script
      google.script.run
        .withSuccessHandler(function (imageUrl) {

          // Si Apps Script devuelve una imagen válida, actualizarla
          if (imageUrl) {

            logo.src = imageUrl;
            localStorage.setItem('cachedLogo', imageUrl);

            // Mostrar logo cuando ya haya cargado
            logo.onload = function () {
              logo.style.opacity = 1;
            };
          }
        })
        .getDriveImage(); // Función en Apps Script que obtiene la imagen desde Google Drive
    });

    /* ---------------------------------------------------
       NAVEGACIÓN ENTRE VISTAS (HTMLs cargados desde Apps Script)
    ---------------------------------------------------- */
    function goTo(view) {
      google.script.run
        .withSuccessHandler(function (html) {
          document.open();
          document.write(html);
          document.close();
        })
        .loadView(view); // Función Apps Script que devuelve el HTML solicitado
    }

    /* Volver a la vista inicial */
    function goBack() {
      google.script.run
        .withSuccessHandler(function (html) {
          document.open();
          document.write(html);
          document.close();
        })
        .loadView(''); // Vista por defecto (menú principal)
    }

  </script>
