  <div class="card">

  <script>
    /* ------------------------------
       VARIABLES GLOBALES
    ------------------------------ */

    let stream; // Flujo de la cámara

    const video = document.getElementById("video");
    const canvas = document.getElementById("canvas");
    const ctx = canvas.getContext("2d");

    const statusEl = document.getElementById("status");
    const progressContainer = document.getElementById("progressContainer");
    const progressBar = document.getElementById("progressBar");
    const errorLog = document.getElementById("errorLog");

    /* ---------------------------------
       INICIO Y DETENCIÓN DE LA CÁMARA
    ---------------------------------- */

    // Inicializa la cámara
    async function startCamera() {
      try {
        stream = await navigator.mediaDevices.getUserMedia({ video: true });
        video.srcObject = stream;
      } catch {
        statusEl.innerText = "No se pudo acceder a la cámara.";
      }
    }

    // Detiene la cámara
    function stopCamera() {
      if (stream) {
        stream.getTracks().forEach(t => t.stop());
      }
    }

    // Activar la cámara automáticamente
    startCamera();

    /* ---------------------------------
       FUNCIÓN PRINCIPAL: REGISTRAR SALIDA
    ---------------------------------- */
    function registrarSalida() {

      // Obtener valores ingresados
      const cedulas = document.getElementById("cedulas").value.trim();
      const obsSalida = document.getElementById("obsSalida").value.trim();
      const btnSalida = document.getElementById("btnSalida");

      // Limpiar errores anteriores
      errorLog.style.display = "none";
      errorLog.innerHTML = "";

      // Validar que se ingresen cédulas
      if (!cedulas) {
        statusEl.innerText = "Debe ingresar al menos una cédula.";
        return;
      }

      // Convertir cédulas separadas por espacio o coma en una lista
      const listaCedulas = cedulas.split(/[\s,]+/).filter(Boolean);

      // Mostrar barra de progreso
      progressContainer.style.display = "block";
      progressBar.style.width = "0%";

      statusEl.innerText = "Registrando salidas...";
      btnSalida.disabled = true;

      /* -------------------------------------------
         OBTENER UBICACIÓN Y CAPTURAR FOTO
      -------------------------------------------- */
      navigator.geolocation.getCurrentPosition(pos => {

        // Ajustar lienzo al tamaño del video
        canvas.width = video.videoWidth;
        canvas.height = video.videoHeight;

        // Tomar foto desde el video
        ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
        const photo = canvas.toDataURL("image/jpeg");

        let procesadas = 0;
        let exitosas = 0;
        let errores = [];

        /* -------------------------------------------
           PROCESAR UNA A UNA LAS CÉDULAS
        -------------------------------------------- */
        listaCedulas.forEach(id => {

          const payload = {
            user_id: id,
            type: "Salida",
            lat: pos.coords.latitude,
            lng: pos.coords.longitude,
            photoDataURL: photo,
            obsSalida: obsSalida
          };

          // Enviar datos a Apps Script
          google.script.run
            .withSuccessHandler(res => {

              procesadas++;

              if (res.success) {
                exitosas++;
              } else {
                errores.push(res.message);
              }

              // Actualizar barra de progreso
              const progreso = Math.round((procesadas / listaCedulas.length) * 100);
              progressBar.style.width = progreso + "%";

              // Si ya procesó todas las cédulas
              if (procesadas === listaCedulas.length) {

                btnSalida.disabled = false;
                statusEl.innerText = `Registros exitosos: ${exitosas}`;

                if (errores.length > 0) {
                  errorLog.style.display = "block";
                  errorLog.innerHTML = errores.join("<br>");
                  statusEl.innerText += "\nAlgunas cédulas presentan errores.";
                } else {
                  statusEl.innerText += "\nTodos los registros fueron exitosos.";
                  setTimeout(() => {
                    stopCamera();
                    goBack();
                  }, 2000);
                }
              }
            })

            .withFailureHandler(err => {
              procesadas++;

              errores.push("Error en cédula " + id + ": " + err.message);

              const progreso = Math.round((procesadas / listaCedulas.length) * 100);
              progressBar.style.width = progreso + "%";

              if (procesadas === listaCedulas.length) {
                btnSalida.disabled = false;

                statusEl.innerText =
                  `Registros exitosos: ${exitosas}/${listaCedulas.length}\n` +
                  "Se presentaron errores en el registro.";

                errorLog.style.display = "block";
                errorLog.innerHTML = errores.join("<br>");
              }
            })

            .saveAttendance(payload);
        });

      }, () => {
        statusEl.innerText = "No se pudo obtener la ubicación.";
      });
    }

    /* ---------------------------------
       Volver al menú principal
    ---------------------------------- */
    function goBack() {
      stopCamera();
      google.script.run.withSuccessHandler(html => {
        document.open();
        document.write(html);
        document.close();
      }).loadView('EmployeeMenu');
    }
  </script>
