    <button onclick="goToView('employee')">Empleado</button>
    <button onclick="goToView('contractor')">Contratista</button>

  <!-- ---------------------------------------------------------
       JAVASCRIPT DE CARGA DE LOGO Y NAVEGACIÓN
  --------------------------------------------------------- -->
  <script>

    /* ---------------------------------------------------------
       CARGAR LOGO DESDE GOOGLE DRIVE Y GUARDARLO EN CACHÉ
    --------------------------------------------------------- */
    document.addEventListener('DOMContentLoaded', function() {
      const logo = document.getElementById('Logo');
      const cachedLogo = localStorage.getItem('cachedLogo');

      // Si existe en caché, mostrarlo de inmediato
      if (cachedLogo) {
        logo.src = cachedLogo;
        logo.style.opacity = 1;
      }

      // Solicitar imagen actualizada desde Google Apps Script
      google.script.run
        .withSuccessHandler(function(imageUrl) {
          if (imageUrl) {
            logo.src = imageUrl;
            localStorage.setItem('cachedLogo', imageUrl);
            logo.onload = () => { logo.style.opacity = 1; };
          }
        })
        .getDriveImage();
    });


    /* ---------------------------------------------------------
       CAMBIAR DE VISTA DINÁMICAMENTE (Employee / Contractor / Admin)
    --------------------------------------------------------- */
    function goToView(view) {
      // Actualiza la URL para permitir navegación hacia atrás
      google.script.history.push({
        state: { view: view },
        location: '?view=' + view
      });

      // Carga la vista solicitada desde el servidor App Script
      google.script.run
        .withSuccessHandler(loadHtml)
        .loadView(view);
    }


    /* Inserta el HTML recibido en toda la página */
    function loadHtml(html) {
      document.open();
      document.write(html);
      document.close();
    }

  </script>
