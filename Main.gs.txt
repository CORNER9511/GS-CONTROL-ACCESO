/******************** INTERFAZ *********************/
function doGet(e) {
  const view = e.parameter.view || '';

  switch (view) {
    case 'employee':
      return HtmlService.createTemplateFromFile('EmployeeMenu').evaluate().setTitle('Menú Empleado');
    case 'EmployeeEntrada':
      return HtmlService.createTemplateFromFile('EmployeeEntrada').evaluate().setTitle('Entrada Empleado');
    case 'EmployeeSalida':
      return HtmlService.createTemplateFromFile('EmployeeSalida').evaluate().setTitle('Salida Empleado');
    case 'contractor':
      return HtmlService.createTemplateFromFile('ContractorMenu').evaluate().setTitle('Menú Contratista');
    case 'ContractorEntrada':
      return HtmlService.createTemplateFromFile('ContractorEntrada').evaluate().setTitle('Entrada Contratista');
    case 'ContractorSalida':
      return HtmlService.createTemplateFromFile('ContractorSalida').evaluate().setTitle('Salida Contratista');
    case 'admin':
      return HtmlService.createTemplateFromFile('Admin').evaluate().setTitle('Menú Administrador');
    default:
      return HtmlService.createTemplateFromFile('Index').evaluate().setTitle('Control de Asistencia');
  }
}

/******************** 📸 Imagen de Drive *********************/
function getDriveImage() {
  const file = DriveApp.getFileById(IMAGE_ID);
  const blob = file.getBlob();
  const contentType = blob.getContentType();
  const bytes = Utilities.base64Encode(blob.getBytes());
  return `data:${contentType};base64,${bytes}`;
}

function doGetImage() {
  return HtmlService.createTemplateFromFile('Imagen')
    .evaluate()
    .setTitle('Imagen desde Drive');
}
